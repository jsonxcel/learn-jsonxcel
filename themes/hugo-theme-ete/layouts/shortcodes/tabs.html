{{- /* Delimiter-based tabs (nested shortcodes are unreliable in Hugo). {{< tabs >}} === curl ```bash curl ... ``` === powershell ```powershell ... ``` {{< /tabs >}} */ -}} {{- $raw := .Inner | strings.TrimSpace -}} {{- $panels := slice -}} {{- if not (hasPrefix $raw "===") -}} {{- $raw = printf "===\n%s" $raw -}} {{- end -}} {{- $chunks := split $raw "\n===" -}} {{- range $chunks -}} {{- $chunk := strings.TrimSpace . -}} {{- if $chunk -}} {{- if hasPrefix $chunk "===" -}} {{- $chunk = strings.TrimSpace (strings.TrimPrefix "===" $chunk) -}} {{- end -}} {{- $lines := split $chunk "\n" -}} {{- $label := index $lines 0 | strings.TrimSpace -}} {{- $body := "" -}} {{- if gt (len $lines) 1 -}} {{- $body = delimit (after 1 $lines) "\n" | strings.TrimSpace -}} {{- end -}} {{- if $label -}} {{- $panels = $panels | append (dict "label" $label "body" $body) -}} {{- end -}} {{- end -}} {{- end -}} {{- if $panels -}}
{{- range $i, $p := $panels -}} {{- end -}}
{{- range $i, $p := $panels -}}

{{- end -}}
